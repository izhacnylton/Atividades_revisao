<script>
    let total = 0;

    function adicionarItem() {
        const select = document.getElementById("selectProduto");
        const prodId = select.value;
        if (!prodId) return alert("Selecione um produto!");

        const option = select.querySelector(`option[value="${prodId}"]`);
        const nome = option.getAttribute("data-nome");
        const preco = parseFloat(option.getAttribute("data-preco"));
        const qtd = parseInt(document.getElementById("inputQtd").value);
        const subtotal = preco * qtd;

        const tbody = document.querySelector("#tabelaItens tbody");
        const row = document.createElement("tr");
        row.innerHTML = `
                <td>${nome} <input type="hidden" name="produtosIds" value="${prodId}"></td>
                <td>${qtd} <input type="hidden" name="quantidades" value="${qtd}"></td>
                <td>R$ ${preco.toFixed(2)}</td>
                <td class="fw-bold">R$ ${subtotal.toFixed(2)}</td>
                <td><button type="button" class="btn btn-outline-danger btn-sm" onclick="removerItem(this, ${subtotal})"><i class="bi bi-x"></i></button></td>
            `;
        tbody.appendChild(row);

        total += subtotal;
        document.getElementById("labelTotal").innerText = "R$ " + total.toFixed(2);
    }

    function removerItem(btn, subtotal) {
        btn.closest("tr").remove();
        total -= subtotal;
        document.getElementById("labelTotal").innerText = "R$ " + total.toFixed(2);
    }
</script>
</body>
</html>
